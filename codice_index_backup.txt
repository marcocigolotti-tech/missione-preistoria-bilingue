function choose(i){if(answered)return;answered=true;const q=game[pos],chosen=currentOptions[i],buttons=[...document.querySelectorAll('.answer')];buttons.forEach((b,n)=>{b.disabled=true;if(currentOptions[n].correct)b.classList.add('correct')});const ok=chosen.correct;if(ok){points++;$('#score').textContent=points}else{buttons[i].classList.add('wrong');errors.push({q,chosen:chosen.text,chosenAr:chosen.ar})}$('#card').classList.add('answered');$('#captionIt').textContent=IMG[q.img][1];$('#captionAr').textContent=IMG_AR[q.img];const f=$('#feedback');f.className='feedback '+(ok?'ok':'no');f.innerHTML=`<div class="status-pair bi-row"><span class="it-col">${ok?'✓ Risposta corretta!':'✗ Non è la risposta giusta.'}</span><span class="ar-col" lang="ar" dir="rtl">${ok?'إجابة صحيحة! ✓':'هذه ليست الإجابة الصحيحة. ✗'}</span></div><div class="bi-row"><p class="it-col">${q.e}</p><p class="ar-col" lang="ar" dir="rtl">${q.eAr}</p></div>`;f.hidden=false;$('#nextIt').textContent=pos===game.length-1?'Vedi il risultato →':'Prossima domanda →';$('#nextAr').textContent=pos===game.length-1?'شاهد النتيجة ←':'السؤال التالي ←';$('#next').hidden=false;$('#bar').style.width=`${(pos+1)/game.length*100}%`;f.scrollIntoView({behavior:'smooth',block:'nearest'})}
function next(){stopSpeech();if(pos<game.length-1){pos++;render();scrollTo({top:0,behavior:'smooth'})}else finish()}
function hint(){const q=game[pos],h=$('#hint');h.innerHTML=`<div class="bi-row"><div class="it-col"><strong>Indizio:</strong> ${q.h}</div><div class="ar-col" lang="ar" dir="rtl"><strong>تلميح:</strong> ${q.hAr}</div></div>`;h.hidden=!h.hidden;$('#hintBtn').setAttribute('aria-expanded',String(!h.hidden))}
let availableVoices=[];
function voiceScore(v,lang){let n=v.name.toLowerCase(),score=0;if(v.lang.toLowerCase().startsWith(lang))score+=100;if(/natural|premium|enhanced|google|microsoft/.test(n))score+=50;if(/elsa|isabella|diego|cosimo|alice|federica|hamed|zeina|laila|salma|majed|zariyah/.test(n))score+=25;return score}
function fillVoiceSelect(id,lang,fallback){const select=$(id),old=select.value,list=availableVoices.filter(v=>v.lang.toLowerCase().startsWith(lang)).sort((a,b)=>voiceScore(b,lang)-voiceScore(a,lang)||a.name.localeCompare(b.name));select.innerHTML=`<option value="">${fallback}</option>`;list.forEach(v=>{const o=document.createElement('option');o.value=v.voiceURI;o.textContent=`${v.name} (${v.lang})`;select.appendChild(o)});if([...select.options].some(o=>o.value===old))select.value=old;else if(list.length)select.value=list[0].voiceURI}
function loadVoices(){if(!('speechSynthesis' in window))return;availableVoices=window.speechSynthesis.getVoices();fillVoiceSelect('#voiceIt','it','Voce automatica del browser');fillVoiceSelect('#voiceAr','ar','صوت المتصفح التلقائي')}
function speechText(lang){const q=game[pos],letter=i=>String.fromCharCode(65+i);return lang==='it'?`${q.q}. ${currentOptions.map((op,i)=>`${letter(i)}. ${op.text}`).join('. ')}`:`${q.qAr}. ${currentOptions.map((op,i)=>`${letter(i)}. ${op.ar}`).join('. ')}`}
function speakLanguage(lang){if(!('speechSynthesis' in window)){alert('La lettura vocale non è disponibile in questo browser.\nالقراءة الصوتية غير متاحة في هذا المتصفح.');return}stopSpeech();const utterance=new SpeechSynthesisUtterance(speechText(lang)),select=$(lang==='it'?'#voiceIt':'#voiceAr'),voice=availableVoices.find(v=>v.voiceURI===select.value);utterance.lang=lang==='it'?'it-IT':'ar-SA';utterance.rate=lang==='it'?.92:.88;utterance.pitch=1;if(voice)utterance.voice=voice;window.speechSynthesis.speak(utterance)}
function finish(){stopSpeech();show('#result');const pct=Math.round(points/game.length*100),name=$('#player').value.trim();$('#finalScore').textContent=`${points}/${game.length}`;$('#resultTitleIt').textContent=name?`Ottimo lavoro, ${name}!`:'Missione completata!';$('#resultTitleAr').textContent=name?`عمل رائع، ${name}!`:'اكتملت المهمة!';const messages=pct>=85?['Conosci molto bene tempi, tecniche e monumenti della Preistoria.','أنت تعرف جيدًا عصور ما قبل التاريخ وتقنياته وآثاره.']:pct>=60?['Buon risultato: rivedi gli indizi delle risposte sbagliate per consolidare.','نتيجة جيدة: راجع تلميحات الإجابات الخاطئة لتثبيت ما تعلمته.']:['Ogni errore è una traccia utile: rivedilo e prova una nuova missione.','كل خطأ أثر مفيد: راجعه وجرّب مهمة جديدة.'];$('#resultText').innerHTML=`<p class="it-col">${messages[0]}</p><p class="ar-col" lang="ar" dir="rtl">${messages[1]}</p>`;$('#reviewBtn').hidden=errors.length===0;$('#review').hidden=true}
function review(){const out=$('#reviewList');out.innerHTML=errors.map(x=>`<article class="review-item"><div class="bi-row"><div class="it-col"><h3>${x.q.q}</h3><div class="mine"><strong>Hai scelto:</strong> ${x.chosen}</div><div class="right"><strong>Risposta corretta:</strong> ${x.q.o[x.q.a]}</div><p>${x.q.e}</p></div><div class="ar-col" lang="ar" dir="rtl"><h3>${x.q.qAr}</h3><div class="mine"><strong>اخترت:</strong> ${x.chosenAr}</div><div class="right"><strong>الإجابة الصحيحة:</strong> ${x.q.oAr[x.q.a]}</div><p>${x.q.eAr}</p></div></div></article>`).join('');$('#review').hidden=false;$('#review').scrollIntoView({behavior:'smooth'})}
